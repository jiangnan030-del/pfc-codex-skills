model new
model domain extent -20.0 20.0 condition periodic
model cmat default model linear property fric 0.25 method deformability emod 1e6 kratio 1.25 

model random 10022 

[rad = 0.5]
[vc   = (4.0/3.0)*math.pi*(rad)^3]
[moic = (2.0/5.0) * vc * rad^2]
clump template create name 'single'                   ...
            pebbles 1                               ...
            @rad 0 0 0                              ...
            volume @vc                              ...
            inertia @moic @moic @moic 0 0 0
model save '1.f3sav'

clump template create name 'dyad'                     ...
            pebbles 2                               ...
            @rad [-rad*0.5] 0 0                     ...
            @rad [rad*0.5] 0 0                      ...
            pebcalculate 0.005
model save '2.f3sav'

geometry import '11.2.6.stl'
clump template create name 'dolos'                    ...
            geometry '11'                          ...
            bubblepack ratio 0.3 distance 120       ...
            surfcalculate
model save '3.f3sav'

clump generate diameter size 1.5 number 50          ...
            box -5.0 5.0 -5.0 5.0 -5.0 0.0          ...
            group 'bottom'
model save 'bottom.f3sav'

model restore '3.f3sav'
clump generate diameter size 1.5 number 25          ...
            box -5.0 5.0 -5.0 5.0 0.0 5.0           ... 
            templates 2                             ... 
            'dyad' 0.3 'dolos' 0.7                      ...
            azimuth 45.0 45.0                       ...
            tilt 90.0 90.0                          ...
            elevation 45.0 45.0                     ...
            group 'top1'
            
clump generate diameter size 1.5 number 25          ...
            box -5.0 5.0 -5.0 5.0 0.0 5.0           ... 
            templates 2                             ... 
            'dyad' 0.7 'dolos' 0.3                      ...
            azimuth -45.0 -45.0                     ...
            tilt 90.0 90.0                          ...
            elevation 45.0 45.0                     ...
            group 'top2'


model solve time 10.0
model save '11.2.6.f3sav'
